    function showAlert(message, type = 'error') {
      const alertBox = document.getElementById('statusAlert');
      alertBox.className = `alert alert-${type}`;
      alertBox.innerText = message;
      alertBox.style.display = 'block';

    function resetAlert() {
      const alertBox = document.getElementById('statusAlert');
      alertBox.style.display = 'none';
      alertBox.innerText = '';

    function resetPreviews() {
      if (currentObjectUrl) {
        URL.revokeObjectURL(currentObjectUrl);
        currentObjectUrl = null;
      }
      document.getElementById('previewContainer').style.display = 'none';
      document.getElementById('previewImage').style.display = 'none';
      document.getElementById('previewVideo').style.display = 'none';
      document.getElementById('previewDoc').style.display = 'none';
      document.getElementById('previewImage').src = '';
      document.getElementById('videoSource').src = '';
    }

    function handleFileSelection(event) {
      resetAlert();
      const file = event.target.files[0];
      if (!file) return;

      selectedFile = file;
      document.getElementById('pUrlMediaDirect').value = '';
      document.getElementById('finalMediaUrlHolder').value = '';
      resetPreviews();

      document.getElementById('previewContainer').style.display = 'block';
      const holderType = document.getElementById('mediaTypeHolder');

      currentObjectUrl = URL.createObjectURL(file);

      if (file.type.startsWith('image/')) {
        holderType.value = 'image';
        const img = document.getElementById('previewImage');
        img.src = currentObjectUrl;
        img.style.display = 'block';
      } else if (file.type.startsWith('video/')) {
        holderType.value = 'video';
        const vid = document.getElementById('previewVideo');
        const src = document.getElementById('videoSource');
        src.src = currentObjectUrl;
        vid.load();
        vid.style.display = 'block';
      } else {
        holderType.value = 'document';
        document.getElementById('docName').innerText = file.name;
        document.getElementById('previewDoc').style.display = 'block';
      }
    }

    function handleUrlInput(url) {
      resetAlert();
      url = url.trim();
      selectedFile = null;
      resetPreviews();

      if (!url) return;

      document.getElementById('previewContainer').style.display = 'block';
      document.getElementById('finalMediaUrlHolder').value = url;
      const holderType = document.getElementById('mediaTypeHolder');

      if (url.match(/\.(mp4|webm|ogg)($|\?)/i)) {
        holderType.value = 'video';
        const vid = document.getElementById('previewVideo');
        document.getElementById('videoSource').src = url;
        vid.load();
        vid.style.display = 'block';
      } else if (url.match(/\.(pdf|doc|docx)($|\?)/i)) {
        holderType.value = 'document';
        document.getElementById('docName').innerText = url.split('/').pop().split('?')[0];
        document.getElementById('previewDoc').style.display = 'block';
      } else {
        holderType.value = 'image';
        const img = document.getElementById('previewImage');
        img.src = url;
        img.style.display = 'block';
      }
    }

      const ext = filename.substring(filename.lastIndexOf('.'));
      const name = filename.substring(0, filename.lastIndexOf('.'));
      const sanitized = name.normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/[^a-zA-Z0-9]/g, "_");
      return `${sanitized}_${Date.now()}${ext}`;
    }

    async function publierRessource(event) {
      event.preventDefault();
      resetAlert();

      const form = document.getElementById('formStudio');
      if (!form.checkValidity()) {
        form.reportValidity();
        return;
      }

      const btnSubmit = document.getElementById('btnSubmitPublish');
      const progressWrapper = document.getElementById('progressWrapper');
      const progressBar = document.getElementById('progressBar');
      const progressText = document.getElementById('progressText');

      btnSubmit.disabled = true;
      btnSubmit.innerHTML = `<i class="fa-solid fa-spinner fa-spin"></i> Publication en cours...`;

      try {
        let finalUrl = document.getElementById('finalMediaUrlHolder').value;
        const mediaType = document.getElementById('mediaTypeHolder').value;

        if (selectedFile) {
          progressWrapper.style.display = 'block';
          progressBar.style.width = '0%';
          progressText.innerText = 'Initialisation...';

          let folderName = 'images';
          if (mediaType === 'video') folderName = 'videos';
          if (mediaType === 'document') folderName = 'documents';

          const sanitizedName = cleanFileName(selectedFile.name);
          const filePath = `${folderName}/${sanitizedName}`;
          const storageRef = storage.ref().child(filePath);
          
          const uploadTask = storageRef.put(selectedFile);

          finalUrl = await new Promise((resolve, reject) => {
            uploadTask.on(
              firebase.storage.TaskEvent.STATE_CHANGED,
              (snapshot) => {
                if (snapshot.totalBytes > 0) {
                  const progress = (snapshot.bytesTransferred / snapshot.totalBytes) * 100;
                  progressBar.style.width = `${progress.toFixed(0)}%`;
                  progressText.innerText = `Envoi (${mediaType.toUpperCase()}) : ${progress.toFixed(0)}%`;
                }
              },
              (error) => {
                reject(new Error("Échec du téléversement vers Firebase Storage : " + error.message));
              },
              async () => {
                try {
                  const downloadURL = await uploadTask.snapshot.ref.getDownloadURL();
                  resolve(downloadURL);
                } catch (err) {
                  reject(err);
                }
              }
            );
          });
        }

        if (!finalUrl) {
          throw new Error("Veuillez sélectionner un fichier (Image, Vidéo, Document) ou renseigner une URL valide.");
        }

        const currentUser = firebase.auth().currentUser;
        const vendorUid = currentUser ? currentUser.uid : "bi7zhxqoYfY3TyZ7dMJuj2yhsx82";

        const maintenant = new Date();
        const dateLisible = maintenant.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit', year: 'numeric' });
        const heureLisible = maintenant.toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit', second: '2-digit' });

        const newRef = db.ref('publications').push();
        const pubData = {
          id: newRef.key,
          nom: document.getElementById('pNom').value.trim(),
          auteur: document.getElementById('pAuteur').value.trim(),
          vendeurNom: document.getElementById('pAuteur').value.trim(),
          vendeurPhone: document.getElementById('pVendeurPhone').value.trim(),
          vendeurUid: vendorUid,
          createdBy: vendorUid,
          categorie: document.getElementById('pCategorie').value,
          type: document.getElementById('pType').value,
          prix: Number(document.getElementById('pPrixPromo').value) || Number(document.getElementById('pPrixNormal').value),
          prixNormal: Number(document.getElementById('pPrixNormal').value),
          prixPromo: Number(document.getElementById('pPrixPromo').value) || Number(document.getElementById('pPrixNormal').value),
          stock: Number(document.getElementById('pStock').value) || 0,
          description: document.getElementById('pDescription').value.trim(),
          image: finalUrl,
          imageUrl: finalUrl,
          mediaUrl: finalUrl,
          mediaType: mediaType,
          createdAt: maintenant.getTime(),
          datePublication: dateLisible,
          heurePublication: heureLisible,
          dateEtHeure: `${dateLisible} à ${heureLisible}`
        };

        await newRef.set(pubData);

        showAlert(`Publication réussie avec succès !`, 'success');
        form.reset();
        resetPreviews();
        selectedFile = null;
        progressWrapper.style.display = 'none';
        switchStep(1);

      } catch (err) {
        showAlert(err.message, 'error');
      } finally {
        btnSubmit.disabled = false;
        btnSubmit.innerHTML = `<i class="fa-solid fa-cloud-arrow-up"></i> Publier maintenant`;
      }
    }
  </script>
