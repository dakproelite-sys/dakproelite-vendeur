  <script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-app.js";
    import { getDatabase, ref, onValue, push } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-database.js";

    const firebaseConfig = {
      databaseURL: "https://votre-projet-default-rtdb.firebaseio.com"
    };

    const app = initializeApp(firebaseConfig);
    const db = getDatabase(app);
    const produitsRef = ref(db, 'produits_vendeurs');

    let listeProduits = [];

    onValue(produitsRef, (snapshot) => {
      listeProduits = [];
      const data = snapshot.val();
      if (data) {
        Object.keys(data).forEach(key => {
          listeProduits.push({ id: key, ...data[key] });
        });
      }
      afficherProduits(listeProduits);
    });

    window.afficherProduits = function(produits) {
      const container = document.getElementById('gridProduits');
      container.innerHTML = '';

      if (produits.length === 0) {
        container.innerHTML = `
          <div class="col-span-full py-16 text-center glass-card rounded-2xl border border-slate-800">
            <div class="w-12 h-12 rounded-xl bg-blue-500/10 border border-blue-500/30 flex items-center justify-center text-blue-400 mx-auto mb-3">
              <svg class="w-6 h-6" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="1.5" d="M20 13V6a2 2 0 00-2-2H6a2 2 0 00-2 2v7m16 0v5a2 2 0 01-2 2H6a2 2 0 01-2-2v-5m16 0h-2.586a1 1 0 00-.707.293l-2.414 2.414a1 1 0 01-.707.293h-3.172a1 1 0 01-.707-.293l-2.414-2.414A1 1 0 006.586 13H4"/></svg>
            </div>
            <p class="text-sm font-semibold text-slate-300">Aucun produit disponible pour le moment.</p>
            <p class="text-xs text-slate-500 mt-1">Cliquez sur "Publier Produit" ou "Devenir Vendeur" pour alimenter le catalogue.</p>
          </div>`;
        return;
      }

      produits.forEach(p => {
        const reduction = (p.prixNormal && p.prixNormal > p.prixPromo) 
          ? Math.round(((p.prixNormal - p.prixPromo) / p.prixNormal) * 100) 
          : 0;
        
        container.innerHTML += `
          <div class="glass-card rounded-2xl overflow-hidden hover:border-slate-700/80 transition-all duration-300 flex flex-col justify-between group">
            <div>
              <div class="relative h-52 bg-slate-950 overflow-hidden">
                <img src="${p.imageUrl}" alt="${p.nom}" class="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500" onerror="this.src='https://via.placeholder.com/400x300/0f172a/94a3b8?text=DAKPROELITE'">
                ${reduction > 0 ? `<span class="absolute top-3 left-3 bg-red-600 text-white text-[11px] font-black px-2.5 py-1 rounded-lg shadow-lg shadow-red-600/30">-${reduction}% OFF</span>` : ''}
                <span class="absolute top-3 right-3 bg-blue-600/90 backdrop-blur-md text-white text-[10px] font-bold uppercase px-2.5 py-1 rounded-lg shadow-md">${p.categorie || 'GENERAL'}</span>
              </div>

              <div class="p-4">
                <h2 class="text-base font-bold text-white truncate">${p.nom}</h2>
                <p class="text-xs text-slate-400 mt-1 line-clamp-2 leading-relaxed">${p.description}</p>
              </div>
            </div>

            <div class="px-4 pb-4 pt-3 border-t border-slate-800/80">
              <div class="flex items-baseline justify-between mb-3">
                <div>
                  <span class="text-xs text-slate-500 block">Prix Revente :</span>
                  <span class="text-lg font-extrabold text-amber-400">${Number(p.prixPromo).toLocaleString('fr-FR')} FCFA</span>
                </div>
                <div class="text-right">
                  <span class="text-xs text-slate-500 block">Prix Conseillé :</span>
                  <span class="text-xs font-semibold text-red-400 line-through">${Number(p.prixNormal).toLocaleString('fr-FR')} FCFA</span>
                </div>
              </div>

              <button onclick="copierPartageSocial('${p.id}')" class="w-full py-2.5 bg-emerald-600/20 hover:bg-emerald-600 text-emerald-400 hover:text-white border border-emerald-500/40 text-xs font-bold rounded-xl transition-all flex items-center justify-center space-x-2 active:scale-95 shadow-md">
                <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M8 16H6a2 2 0 01-2-2V6a2 2 0 012-2h8a2 2 0 012 2v2m-6 12h8a2 2 0 002-2v-8a2 2 0 00-2-2h-8a2 2 0 00-2 2v8a2 2 0 002 2z"/></svg>
                <span>Copier Texte de Vente</span>
              </button>
            </div>
          </div>
        `;
      });
    };

    window.sauvegarderProduit = function(e) {
      e.preventDefault();
      const nouveauProduit = {
        nom: document.getElementById('nom').value,
        prixPromo: Number(document.getElementById('prixPromo').value),
        prixNormal: Number(document.getElementById('prixNormal').value),
        categorie: document.getElementById('categorie').value || 'Général',
        imageUrl: document.getElementById('imageUrl').value,
        description: document.getElementById('description').value,
        dateCreation: Date.now()
      };

      push(produitsRef, nouveauProduit).then(() => {
        fermerModalAjout();
        document.getElementById('formProduit').reset();
        mettreAJourApercu();
        afficherToast("Produit publié avec succès !");
      });
    };

    window.copierPartageSocial = function(id) {
      const p = listeProduits.find(item => item.id === id);
      if (!p) return;

      const texteSocial = 
`🔥 *OFFRE SPECIALE - DISPONIBLE EN STOCK* 🔥
📦 *${p.nom.toUpperCase()}*

📝 ${p.description}

💰 *Prix Promo : ${Number(p.prixPromo).toLocaleString('fr-FR')} FCFA*
❌ Prix Habituel : ~${Number(p.prixNormal).toLocaleString('fr-FR')} FCFA~

📦 *Livraison rapide disponible partout*
📲 Contactez-nous pour commander !`;

      navigator.clipboard.writeText(texteSocial).then(() => {
        afficherToast("Texte de revente copié ! Collez-le sur WhatsApp / Facebook.");
      });
    };

    window.copierToutLeCatalogue = function() {
      if (listeProduits.length === 0) return;

      let texteCatalogue = `💥 *CATALOGUE OFFRES DROPSHIPPING - DAKPROELITE* 💥\n\n`;
      
      listeProduits.forEach((p, idx) => {
        texteCatalogue += `${idx + 1}. *${p.nom}*\n💰 ${Number(p.prixPromo).toLocaleString('fr-FR')} FCFA (au lieu de ~${Number(p.prixNormal).toLocaleString('fr-FR')} FCFA~)\n\n`;
      });

      texteCatalogue += `📲 *Contactez-moi directement pour passer votre commande !*`;

      navigator.clipboard.writeText(texteCatalogue).then(() => {
        afficherToast("Catalogue complet copié ! Prêt pour vos canaux de revente.");
      });
    };
  </script>

  <!-- Script UI & Génération de lien Vendeur Personnalisé -->
  <script>
    function ouvrirModalAjout() { 
      document.getElementById('modalAjout').classList.remove('hidden'); 
      mettreAJourApercu();
    }

    function fermerModalAjout() { 
      document.getElementById('modalAjout').classList.add('hidden'); 
    }

    function afficherToast(message) {
      const toast = document.getElementById('toastNotification');
      document.getElementById('toastMessage').innerText = message;
      toast.classList.remove('hidden');
      setTimeout(() => { toast.classList.add('hidden'); }, 3000);
    }

    // Générer et copier un lien vendeur au nom du revendeur
    function copierMonLienVendeur() {
      const nomVendeur = prompt("Entrez votre Nom / Identifiant Vendeur :", "mon_nom");
      if (!nomVendeur) return;

      const baseDomain = window.location.origin; // Récupère le nom de domaine de votre application
      const slug = nomVendeur.toLowerCase().trim().replace(/[^a-z0-9]/g, '_');
      const lienVendeur = `${baseDomain}/vendeur.html?ref=${slug}`;

      navigator.clipboard.writeText(lienVendeur).then(() => {
        afficherToast(`Lien généré et copié : ${lienVendeur}`);
      });
    }

    // Mise à jour instantanée de l'aperçu
    function mettreAJourApercu() {
      const nom = document.getElementById('nom').value || 'Titre du produit';
      const prixPromo = Number(document.getElementById('prixPromo').value) || 0;
      const prixNormal = Number(document.getElementById('prixNormal').value) || 0;
      const categorie = document.getElementById('categorie').value || 'CATEGORIE';
      const imageUrl = document.getElementById('imageUrl').value;
      const description = document.getElementById('description').value || 'La description du produit s\'affichera ici au fur et à mesure de votre saisie.';

      document.getElementById('prevNom').innerText = nom;
      document.getElementById('prevDesc').innerText = description;
      document.getElementById('prevCategorie').innerText = categorie;
